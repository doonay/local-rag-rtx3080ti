import pytest
from pathlib import Path
from document_processor.parsers import parse_pdf, parse_docx, parse_html

class TestParsers:
    def test_parse_pdf(self, sample_pdf_path):
        """Тестируем парсинг нормального PDF"""
        with open(sample_pdf_path, 'rb') as f:
            pdf_bytes = f.read()
        
        text = parse_pdf(pdf_bytes)
        
        assert isinstance(text, str)
        assert len(text) > 0
        assert "test" in text.lower() or "тест" in text.lower()
    
    def test_parse_pdf_empty(self):
        """Тестируем обработку пустого PDF"""
        with pytest.raises(ValueError, match="Empty PDF"):
            parse_pdf(b'')
    
    def test_parse_docx(self, sample_docx_path):
        with open(sample_docx_path, 'rb') as f:
            docx_bytes = f.read()
        
        text = parse_docx(docx_bytes)
        assert isinstance(text, str)
        assert len(text) > 0
    
    def test_parse_html(self):
        html = "<html><body><h1>Title</h1><p>Content</p></body></html>"
        text = parse_html(html.encode())
        
        assert "Title" in text
        assert "Content" in text

import pytest
from document_processor.cleaner import clean_text

class TestCleaner:
    def test_remove_extra_newlines(self):
        text = "Line 1\n\n\n\nLine 2"
        cleaned = clean_text(text)
        assert "\n\n" not in cleaned
    
    def test_remove_double_spaces(self):
        text = "This  has   multiple    spaces"
        cleaned = clean_text(text)
        assert "  " not in cleaned
    
    def test_normalize_unicode(self):
        text = "Café résumé naïve"
        cleaned = clean_text(text)
        # Проверяем, что символы нормализованы (но не потеряны)
        assert "é" in cleaned or "e" in cleaned
    
    def test_remove_page_numbers(self):
        text = "Some text\nPage 1\nMore text\nPage 2"
        cleaned = clean_text(text)
        assert "Page 1" not in cleaned
        assert "Page 2" not in cleaned
    
    def test_remove_headers_footers(self):
        text = "Header: Confidential\nContent\nFooter: Page 1"
        cleaned = clean_text(text)
        assert "Header:" not in cleaned
        assert "Footer:" not in cleaned
    
    def test_full_pipeline(self):
        raw = """Page 1
        This is a  test   document.
        Header: Internal
        
        
        This document contains important information.
        Footer: Draft"""
        
        cleaned = clean_text(raw)
        
        assert "Page 1" not in cleaned
        assert "Header:" not in cleaned
        assert "Footer:" not in cleaned
        assert "  " not in cleaned
        assert "\n\n" not in cleaned
        assert "important" in cleaned

import pytest
from document_processor.chunker import chunk_text

class TestChunker:
    def test_chunk_size_limit(self):
        """Проверяем, что чанки не превышают max_size"""
        text = " ".join(["word"] * 1000)
        chunks = chunk_text(text, chunk_size=100, overlap=20)
        
        for chunk in chunks:
            assert len(chunk) <= 100
    
    def test_chunk_overlap(self):
        """Проверяем наличие перекрытия между чанками"""
        text = " ".join([f"sentence_{i}" for i in range(100)])
        chunks = chunk_text(text, chunk_size=50, overlap=10)
        
        # Первый чанк должен содержать конец, который есть в начале второго
        # Это сложно проверить строго, но можем проверить, что текст не потерян
        full_text = "".join(chunks)
        assert len(full_text) >= len(text) * 0.9  # С учетом перекрытия
    
    def test_chunk_preserves_sentences(self):
        """Чанкер должен стараться не резать посередине предложения"""
        text = "First sentence. Second sentence. Third sentence."
        chunks = chunk_text(text, chunk_size=20, overlap=0)
        
        # Проверяем, что предложения не разорваны посередине
        for chunk in chunks:
            # Каждый чанк должен начинаться с заглавной или заканчиваться точкой
            # Это не строгое требование, но хорошая практика
            pass
    
    def test_empty_text(self):
        text = ""
        chunks = chunk_text(text)
        assert chunks == []
    
    def test_short_text(self):
        text = "Short text"
        chunks = chunk_text(text, chunk_size=100)
        assert len(chunks) == 1
        assert chunks[0] == text

            host="localhost",
            port=6333,
            collection_name="test_collection"
        )

        return [
            DocumentChunk(
                id="1",
                text="Paris is the capital of France",
                metadata={"source": "test.pdf", "page": 1}
            ),
            DocumentChunk(
                id="2",
                text="Berlin is the capital of Germany",
                metadata={"source": "test.pdf", "page": 2}
            )
        ]

        assert success is True

        query_vector = [0.1, 0.2, 0.3]
        results = qdrant_client.search(query_vector, limit=1)
        
        assert len(results) == 1
        assert results[0]["payload"]["text"] == "Paris is the capital of France"
    
    def test_filter_search(self, qdrant_client, sample_chunks):

        qdrant_client.upsert(sample_chunks, embeddings)
        
        # Поиск с фильтром по документу

            query_vector=[0.1, 0.2, 0.3],
            limit=10,
            filter={"document_id": "doc_123"}  # Должен вернуть 0

📊 Запуск тестов
1. Unit-тесты (быстрые, без внешних зависимостей)
pytest api/tests/unit/ document_processor/tests/unit/ -v

2. Integration-тесты (требуют поднятых контейнеров)
# Поднимаем тестовые контейнеры
docker-compose -f docker-compose.test.yml up -d

# Ждем готовности
sleep 10

# Запускаем интеграционные тесты
pytest api/tests/integration/ -v -m integration

# Останавливаем
docker-compose -f docker-compose.test.yml down

3. E2E-тесты (требуют полной системы)
# Вся система должна быть поднята
docker-compose up -d

# Запускаем E2E
pytest api/tests/e2e/ -v -m e2e

🎯 Итоговый план действий
Сначала пишем schemas.py и тесты к нему

Пишем unit-тесты для каждого компонента (Parser → Cleaner → Chunker)

Разрабатываем компоненты, прогоняя тесты

Пишем интеграционные тесты для клиентов (Embedder → Qdrant → Reranker → vLLM)

Разрабатываем клиенты, используя TDD

Пишем E2E тесты для полных пайплайнов

Настраиваем CI/CD для автоматического запуска всех тестов

Важно: Начните с одного маленького компонента (например, Cleaner) и полностью покройте его тестами. Потом переходите к следующему. Так вы сможете двигаться итеративно, не теряя фокус.
